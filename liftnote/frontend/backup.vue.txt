    d = new Date();
  d.setHours(0, 0, 0, 0);
  while (true) {
    const hasSesh = sessions.value.some((sess) => {
      const sd = new Date(sess.started_at);
      sd.setHours(0, 0, 0, 0);
      return sd.getTime() === d.getTime() && sess.status === "completed";
    });
    if (!hasSesh) break;
    s++;
    d.setDate(d.getDate() - 1);
  }
  return s;

const calendarDays = computed(() => {
  const now = new Date();
  const d = new Date(now.getFullYear(), now.getMonth() + calMonth.value, 1);
  const year = d.getFullYear(),
    month = d.getMonth();
  const firstDay = new Date(year, month, 1).getDay();
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const days = [];
  for (let i = 0; i < firstDay; i++) {
    const prev = new Date(year, month, -firstDay + i + 1);
    days.push({
      n: prev.getDate(),
      key: "p" + i,
      inMonth: false,
      isToday: false,
      hasSession: false,
    });
  }
  for (let i = 1; i <= daysInMonth; i++) {
    const thisDate = new Date(year, month, i);
    const isToday = thisDate.toDateString() === now.toDateString();
    const hasSession = sessions.value.some((s) => {
      const sd = new Date(s.started_at);
      return (
        sd.getFullYear() === year &&
        sd.getMonth() === month &&
        sd.getDate() === i &&
        s.status === "completed"
      );
    });
    days.push({ n: i, key: "c" + i, inMonth: true, isToday, hasSession });
  }
  const remaining = 42 - days.length;
  for (let i = 1; i <= remaining; i++) {
    days.push({
      n: i,
      key: "n" + i,
      inMonth: false,
      isToday: false,
      hasSession: false,
    });
  }
  return days;
});

const last28Days = computed(() => {
  const days = [];
  for (let i = 27; i >= 0; i--) {
    const d = new Date();
    d.setDate(d.getDate() - i);
    d.setHours(0, 0, 0, 0);
    const count = sessions.value.filter((s) => {
      const sd = new Date(s.started_at);
      sd.setHours(0, 0, 0, 0);
      return sd.getTime() === d.getTime() && s.status === "completed";
    }).length;
    const label = i % 7 === 0 || i === 0 ? months[d.getMonth()] : "";
    days.push({ count, label });
  }
  return days;
});

// ── Session ──
function startSession(workout) {
  const exs = exercises.value
    .filter((e) => e.workout_id === workout._id)
    .sort((a, b) => a.order - b.order);
  activeSession.value = {
    _id: genId(),
    user_id: currentUser.value._id,
    workout_id: workout._id,
    workoutName: workout.name,
    exercises: exs,
    started_at: new Date().toISOString(),
    status: "in_progress",
    notes: "",
  };
  // Init set logs
  setLogs.value = {};
  exs.forEach((ex) => {
    const sets = getExSets(ex);
    setLogs.value[ex._id] = sets.map((s) => ({
      ...s,
      weightLogged: s.targetWeight,
      repsLogged: s.targetReps,
      done: false,
    }));
  });
  timerSeconds.value = 0;
  startTimer();
  activeExerciseIdx.value = 0;
  navigate("session");
}

function getExSets(ex) {
  if (ex.set_type === "pyramid" && ex.pyramid_sets?.length) {
    return ex.pyramid_sets.map((s) => ({
      targetReps: s.reps,
      targetWeight: s.weight_kg,
    }));
  }
  return Array.from({ length: ex.series || 3 }, () => ({
    targetReps: ex.reps || 10,
    targetWeight: ex.weight_kg || 0,
  }));
}

function logSet(exId, setIdx, field, value) {
  if (!setLogs.value[exId]) return;
  if (field === "weight")
    setLogs.value[exId][setIdx].weightLogged = parseFloat(value) || 0;
  if (field === "reps")
    setLogs.value[exId][setIdx].repsLogged = parseInt(value) || 0;
}

function markSetDone(exId, setIdx) {
  if (!setLogs.value[exId]) return;
  setLogs.value[exId][setIdx].done = true;
}

function exerciseComplete(ex) {
  const logs = setLogs.value[ex._id];
  return logs && logs.length > 0 && logs.every((s) => s.done);
}

  stopTimer();

  else startTimer();

}

async function generateAISummary() {
  if (!selectedSessionForAI.value) return;
  await generateAISummaryForSession(selectedSessionForAI.value);
}

async function generateAISummaryForSession(sess) {
  aiLoading.value = true;
  const wName = getWorkoutName(sess.workout_id);
  const exs = exercises.value.filter((e) => e.workout_id === sess.workout_id);
  const exerciseList = exs
    .map(
      (e) =>
        `${e.name} (${e.muscle_group}, ${e.series || "?"}x${e.reps || "?"} @ ${e.weight_kg || 0}kg)`,
    )
    .join(", ");
  const prompt = `Você é um personal trainer especialista. Analise esta sessão de treino e dê um resumo motivacional e técnico em português:
Treino: ${wName}
Duração: ${sess.duration_formatted || formatDuration(sess.duration_seconds) || "não registrado"}
Volume total: ${sess.totalVolume || 0}kg
Exercícios: ${exerciseList || "não registrados"}
Notas do usuário: ${sess.notes || "nenhuma"}
Responda em 2-3 frases, sendo específico sobre o treino realizado. Inclua uma dica técnica ou motivacional.`;

  try {
    const resp = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        model: "claude-sonnet-4-20250514",
        max_tokens: 300,
        messages: [{ role: "user", content: prompt }],
      }),
    });
    const data = await resp.json();
    const summary =
      data.content?.map((b) => b.text || "").join("") ||
      "Ótimo treino! Continue assim.";
    sess.ai_summary = summary;
    // Update in sessions array
    const idx = sessions.value.findIndex((s) => s._id === sess._id);
    if (idx >= 0) sessions.value[idx].ai_summary = summary;
    if (finishedSession.value?._id === sess._id)
      finishedSession.value.ai_summary = summary;
    if (viewingSession.value?._id === sess._id)
      viewingSession.value.ai_summary = summary;
    if (selectedSessionForAI.value?._id === sess._id)
      selectedSessionForAI.value = {
        ...selectedSessionForAI.value,
        ai_summary: summary,
      };
  } catch (e) {
    sess.ai_summary =
      "Treino concluído! Ótimo trabalho. Continue mantendo a consistência para alcançar seus objetivos.";
  }
  aiLoading.value = false;
}

function navigate(p) {
  if (p === "session" && !activeSession.value) return;
  page.value = p;
  sidebarOpen.value = false;
}

function genId() {
  return Math.random().toString(36).slice(2) + Date.now().toString(36);
}

function getWorkoutName(id) {
  return (
    workouts.value.find((w) => w._id === id)?.name || "Treino desconhecido"
  );
}

function formatDate(d) {
  return new Date(d).toLocaleDateString("pt-BR", {
    day: "2-digit",
    month: "2-digit",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  });
}

function formatDuration(secs) {
  if (!secs) return null;
  const m = Math.floor(secs / 60),
    s = secs % 60;
  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
}

function statusLabel(s) {
  return (
    {
      completed: "Completo",
      in_progress: "Em andamento",
      cancelled: "Cancelado",
    }[s] || s
  );
}
